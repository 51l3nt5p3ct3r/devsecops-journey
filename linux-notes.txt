ls — מציג קבצים בתיקייה הנוכחית  
pwd — מציג את הנתיב הנוכחי  
cd <path> — מעבר בין תיקיות  
mkdir <name> — יצירת תיקייה חדשה  
touch <file> — יצירת קובץ חדש (ב-Windows אפשר גם ni <file>)  
cat <file> — הצגת תוכן של קובץ  
cp <src> <dst> — העתקת קובץ או תיקייה  
mv <src> <dst> — העברת או שינוי שם של קובץ  
rm <file> — מחיקת קובץ  
rm -r <dir> — מחיקת תיקייה  
grep "<text>" <file> — חיפוש טקסט בתוך קובץ  
find . -name "<pattern>" — חיפוש קובץ לפי שם  
chmod +x <file> — שינוי הרשאות (הפיכת קובץ להרצה בלינוקס)  
curl <url> — הורדת תוכן מאתר  
wget <url> — הורדת קובץ מאתר
